مهم:
این فایل یک بسته کد/دستور برای ادغام در پروژه است، نه ادعای آماده بودن APK نهایی. کدهای Room و محدودیت ظرفیت باید با نسخه Kotlin/Room پروژه تطبیق و build شوند. اصلاح اتصال واقعی Xray/V2Ray نیازمند بررسی API واقعی فایل app/libs/libv2ray.aar و آزمایش روی دستگاه است.

package com.example.data.local

import java.security.MessageDigest
import java.util.Locale

/**
 * Fingerprint for deduplication only. Never log the input fields because they may contain secrets.
 * Do not identify a server only by host:port: different credentials/transports may be valid.
 */
object ConfigFingerprint {
    fun of(server: ServerConfigEntity): String {
        val canonical = listOf(
            server.protocol.trim().lowercase(Locale.ROOT),
            server.host.trim().trim('[', ']').lowercase(Locale.ROOT),
            server.port.toString(),
            server.uuidOrPassword.trim(),
            server.flow.trim(),
            server.securityType.trim().lowercase(Locale.ROOT),
            server.transportType.trim().lowercase(Locale.ROOT),
            server.sni.trim().lowercase(Locale.ROOT),
            server.publicKey.trim(),
            server.shortId.trim(),
            server.fingerprint.trim().lowercase(Locale.ROOT),
            server.path.trim(),
            server.serviceName.trim(),
            server.alpn.trim(),
            server.headerType.trim().lowercase(Locale.ROOT),
            server.alterId.toString()
        ).joinToString("\u001f")

        val digest = MessageDigest.getInstance("SHA-256")
            .digest(canonical.toByteArray(Charsets.UTF_8))
        return digest.joinToString("") { byte -> "%02x".format(byte) }
    }
}

هـ) متد insertAll(servers: List<ServerConfigEntity>) فعلی را حذف کن تا مسیرهای جدید نتوانند سقف ۱۲تایی را ناخواسته دور بزنند. همه فراخوانی‌های آن باید به متدهای کنترل‌شده زیر مهاجرت کنند. insertServer را نگه دار چون متد داخلی این توابع است.

و) متد زیر را داخل ServerDao اضافه کن:

@Transaction
suspend fun insertCapped(
    servers: List<ServerConfigEntity>,
    limit: Int = ServerLimits.MAX_ACTIVE_CONFIGS
): Int {
    require(limit in 1..ServerLimits.MAX_ACTIVE_CONFIGS) {
        "Active server limit cannot exceed ${ServerLimits.MAX_ACTIVE_CONFIGS}"
    }

    // Also repairs a previously oversized database before new inserts.
    trimToLimit(limit)

    val seen = getAllServersOnce()
        .mapTo(mutableSetOf()) { ConfigFingerprint.of(it) }

    var count = getServerCountNow()
    var insertedCount = 0

    for (candidate in servers) {
        if (count >= limit) break
        val fingerprint = ConfigFingerprint.of(candidate)
        if (!seen.add(fingerprint)) continue

        // Parsed/imported configs are candidates, not verified working tunnels.
        val unverified = candidate.copy(
            isWorking = false,
            pingMs = -1L,
            lastTestedTime = 0L
        )
        val id = insertServer(unverified)
        if (id != -1L) {
            count++
            insertedCount++
        }
    }
    return insertedCount
}

ز) اگر می‌خواهی جایگزینی اتمیک داشته باشی، این helper فقط برای candidateای استفاده شود که قبلاً با هسته واقعی و عبور ترافیک تأیید شده است؛ ServerProber تنها برای این کار کافی نیست:

@Transaction
suspend fun insertVerifiedReplacingBroken(
    candidate: ServerConfigEntity,
    limit: Int = ServerLimits.MAX_ACTIVE_CONFIGS
): Boolean {
    require(limit in 1..ServerLimits.MAX_ACTIVE_CONFIGS)
    trimToLimit(limit)

    val existing = getAllServersOnce()
    val candidateFingerprint = ConfigFingerprint.of(candidate)
    if (existing.any {
            it.rawLink == candidate.rawLink || ConfigFingerprint.of(it) == candidateFingerprint
        }) return false

    if (getServerCountNow() >= limit) {
        val victimId = getReplaceableBrokenId() ?: return false
        deleteServer(victimId)
    }

    // Contract: call this only after a real core/tunnel HTTP test passed.
    val insertedId = insertServer(
        candidate.copy(isWorking = true, lastTestedTime = System.currentTimeMillis())
    )
    if (insertedId == -1L) {
        // Throwing causes a Room @Transaction to roll back the victim deletion.
        error("Verified candidate insert conflicted; transaction must roll back")
    }
    return true
}

۲. داخل refreshAllSources، این کد فعلی را حذف کن:

تعداد دریافت‌شده را با تعداد ذخیره‌شده اشتباه نگیر. پیام منبع باید مثلاً «۵ کانفیگ جدید ذخیره شد از ۱۲۰ لینک استخراج‌شده» نشان دهد، نه اینکه همه را درج‌شده گزارش کند.

۳. testLatency فعلی نباید مثبت بودن TCP را به معنای سالم بودن پروتکل/تونل تلقی کند. آن را به این شکل تغییر بده:

suspend fun testLatency(server: ServerConfigEntity): Long = withContext(Dispatchers.IO) {
    val ping = ServerProber.probe(server)
    val now = System.currentTimeMillis()
    if (ping > 0L) {
        // Endpoint reachable only; preserve previous full-tunnel validation status.
        db.serverDao().updatePing(server.id, ping, now)
    } else {
        // Endpoint unreachable. Do not let it be selected as the next candidate.
        db.serverDao().updatePingAndStatus(server.id, -1L, false, now)
    }
    ping
}

چون پروژه version = 2 دارد، محدودیت و بازنشانی وضعیت رکوردهای قدیمی باید برای دیتابیس‌های نصب‌شده قبلی هم اعمال شود.

Import:
import androidx.room.migration.Migration

در companion object اضافه کن:
private val MIGRATION_2_3 = object : Migration(2, 3) {
    override fun migrate(db: SupportSQLiteDatabase) {
        // Old rows were often marked working without a real tunnel check.
        db.execSQL("UPDATE server_configs SET isWorking = 0, pingMs = -1, lastTestedTime = 0")
        // Keep at most 12 records; favorites/manual items are considered first.
        db.execSQL("""
            DELETE FROM server_configs
            WHERE id NOT IN (
                SELECT id FROM server_configs
                ORDER BY isFavorite DESC, isManual DESC, createdAt DESC, id ASC
                LIMIT 12
            )
        """.trimIndent())
    }
}

تغییر annotation دیتابیس به:
@Database(..., version = 3, exportSchema = false)

و در Room.databaseBuilder(...):
.addMigrations(MIGRATION_2_3)

پروژه فعلاً fallbackToDestructiveMigration() دارد. بررسی کن مسیرهای migration مورد نیاز نسخه‌های نصب‌شده پوشش داده شوند؛ در غیر این صورت fallback می‌تواند داده‌های کاربر را حذف کند. بعد از تغییر، schema validation و upgrade از دیتابیس نسخه ۲ را واقعاً آزمایش کن.

============================================================
H) مشکل اتصال واقعی: این مورد با تغییر تعداد رکورد حل نمی‌شود
============================================================
در JavadVpnService.kt این خط وجود دارد:
    builder.addDisallowedApplication(packageName)

بنابراین درخواست OkHttp داخل verifyLiveInternet()، چون از UID خود همین برنامه است، می‌تواند از VPN مستثنا و از اینترنت مستقیم گوشی ارسال شود. این تست در شکل فعلی اثبات نمی‌کند که ترافیک از تونل عبور کرده است؛ ممکن است مثبت کاذب یا منفی کاذب ایجاد کند.

نکته ایمنی فنی: این خط را کورکورانه حذف نکن؛ اگر سوکت‌های خروجی هسته protect نشده باشند، ترافیک سرور می‌تواند دوباره وارد TUN شود و loop ایجاد کند.

از Google AI Studio بخواه:
1. API واقعی callbackها و نسخه موجود در app/libs/libv2ray.aar را بررسی کند.
2. مسیر protect سوکت‌های خروجی core را طبق API واقعی همان AAR پیاده کند.
3. تست اتصال را طوری پیاده کند که درخواست آزمایشی قطعاً از outbound پروکسی عبور کند؛ نه از اینترنت مستقیم.
4. اگر SDK امکان تست candidate بدون ساخت تونل یا بدون ایجاد loop را نمی‌دهد، معماری تست را با یک مسیر local SOCKS/HTTP موقت و outbound کنترل‌شده بازطراحی کند.
5. فقط بعد از handshake واقعی و دریافت پاسخ معتبر از درخواست عبوری، recordSuccess و isWorking=true انجام شوند.
6. به باز بودن TCP port اکتفا نکند: ServerProber فعلی نهایتاً endpoint reachability و تلاش TLS عادی را می‌سنجد؛ Reality و احراز هویت VLESS/VMess/Trojan/SS را اثبات نمی‌کند.

همچنین DefaultVpnCoreEngine.establishSession به‌تنهایی فقط یک VpnTunnelSession ساختگی در حافظه می‌سازد و state را Established می‌کند؛ خودش هسته را اجرا نمی‌کند. چون JavadVpnService جداگانه Libv2ray.startLoop را صدا می‌زند، این wrapper نباید منبع اثبات اتصال باشد. وضعیت نهایی را به بررسی موفق واقعی در JavadVpnService گره بزن و از ایجاد دو منبع حقیقت برای state اتصال جلوگیری کن.
